3.  Show the relationships between entities class

    3.1 One Customer → Many Orders (for creating the Customer Class Enities)
        One Customer → Many Orders
        Customer  1 -------- * Order
    Class Customer
    {
        public int Id {get; set;}
        public string Name {get; set;}

        // Navigation between List of Orders
        public List<Order> Orders {get; set;}

    }

    Class Order
    {
        public int Id {get; set;}

        //Foriegn Key
        public int CustomerId {get; set}

         // Navigation property
        public Customer Customer {get; set;}

    }

    3.2 Creating the Order Class Entities (for understaing Created one more Order Enity)
        One Order → Many OrderItems
        Order  1 -------- * OrderItem
    Class Order
    {
         public int Id {get; set;}

         public List<OrderItem> Items {get; set;}
    }

    Class OrderItem
    {
          public int Id { get; set; }
          public int OrderId { get; set; }
          public Order Order {get; set;}
          public int Quantity {get; set;}
    }
